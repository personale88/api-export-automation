Example (Python requests)
import requests

BASE_URL = "http://127.0.0.1:5000"

res = requests.post(f"{BASE_URL}/api/buyers/search", json={"keyword": "Singing Bowls", "max_results": 5}).json()
print("Added:", res.get("new_leads_added"))

buyers = requests.get(f"{BASE_URL}/api/buyers", params={"country": "United States"}).json()
print("Count:", len(buyers.get("buyers", [])))

email_res = requests.post(f"{BASE_URL}/api/email/test", json={"recipient": "user@example.com"}).json()
print("Status:", email_res)
